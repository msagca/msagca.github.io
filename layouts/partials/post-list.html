{{- /* Single-column post list with previews in alternating side columns. Every row has the same fixed height, so an image's size can be decided here at build time, in row units, with no dependence on the viewport. Previews alternate left/right in the order they appear, so an image may extend down until the next preview on the same side begins (or the list ends). Within that limit it takes as many rows as its aspect ratio asks for at the fixed column width. */ -}} {{- $pages := . -}} {{- $cfg := site.Params.postList | default dict -}} {{- $rowH := float ($cfg.rowHeight | default 3) -}} {{- $gap := float ($cfg.rowGap | default 1.25) -}} {{- $imgW := float ($cfg.imageWidth | default 12) -}} {{- $pitch := add $rowH $gap -}} {{- /* Pass 1: resolve each post's preview and its height/width ratio. */ -}} {{- $rows := slice -}} {{- $imageRows := slice -}} {{- range $i, $page := $pages -}} {{- $kind := "" }}{{ $src := "" }}{{ $shader := "" }}{{ $ratio := 0.5625 -}} {{- with $page.Params.shader -}} {{- with $page.Resources.GetMatch . }}{{ $kind = "shader" }}{{ $shader = .Content }}{{ end -}} {{- with $page.Params.shaderRatio }}{{ $ratio = float . }}{{ end -}} {{- else -}} {{- with $page.Params.image -}} {{- $kind = "image" -}} {{- with $page.Resources.GetMatch . -}} {{- $src = .RelPermalink -}} {{- if eq .MediaType.SubType "svg" -}} {{- with findRESubmatch `viewBox="[-\d.]+[ ,]+[-\d.]+[ ,]+([\d.]+)[ ,]+([\d.]+)"` .Content 1 -}} {{- $m := index . 0 }}{{ $ratio = div (float (index $m 2)) (float (index $m 1)) -}} {{- end -}} {{- else if and (eq .ResourceType "image") (gt .Width 0) -}} {{- $ratio = div (float .Height) (float .Width) -}} {{- end -}} {{- else -}} {{- $src = . -}} {{- end -}} {{- with $page.Params.imageRatio }}{{ $ratio = float . }}{{ end -}} {{- else -}} {{- with $page.Params.youtube -}} {{- $kind = "image" }}{{ $src = printf "https://img.youtube.com/vi/%s/hqdefault.jpg" . }}{{ $ratio = 0.75 -}} {{- end -}} {{- end -}} {{- end -}} {{- if $kind }}{{ $imageRows = $imageRows | append $i }}{{ end -}} {{- $rows = $rows | append (dict "page" $page "kind" $kind "src" $src "shader" $shader "ratio" $ratio) -}} {{- end -}} {{- $n := len $rows -}}
